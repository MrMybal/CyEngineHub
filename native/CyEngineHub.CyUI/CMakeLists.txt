cmake_minimum_required(VERSION 3.21)
project(CyEngineHubCyUI LANGUAGES CXX)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
set(CYHUB_CYUI_SOURCE "" CACHE PATH "Pinned CyUI checkout")
set(CYHUB_CIMGUI_SOURCE "" CACHE PATH "Pinned cimgui checkout including its imgui submodule")
if(NOT EXISTS "${CYHUB_CYUI_SOURCE}/docs/MODULES.md" OR NOT EXISTS "${CYHUB_CIMGUI_SOURCE}/imgui/imgui.h")
  message(FATAL_ERROR "Run tools/ui/build-cyui-native.ps1 to prepare the pinned dependencies")
endif()
add_library(cyhub_imgui STATIC
  ${CYHUB_CIMGUI_SOURCE}/imgui/imgui.cpp ${CYHUB_CIMGUI_SOURCE}/imgui/imgui_draw.cpp
  ${CYHUB_CIMGUI_SOURCE}/imgui/imgui_tables.cpp ${CYHUB_CIMGUI_SOURCE}/imgui/imgui_widgets.cpp
  ${CYHUB_CIMGUI_SOURCE}/imgui/imgui_demo.cpp)
target_include_directories(cyhub_imgui PUBLIC ${CYHUB_CIMGUI_SOURCE}/imgui ${CYHUB_CIMGUI_SOURCE})
target_compile_definitions(cyhub_imgui PUBLIC IMGUI_USE_WCHAR32 IMGUI_DISABLE_DEMO_WINDOWS)
set(CYUI_IMGUI_TARGET cyhub_imgui)
set(CYUI_MODULES "DocumentViews;Diagnostics;Inspector;Images")
set(CYUI_BUILD_DEMO OFF CACHE BOOL "" FORCE)
set(CYUI_BUILD_AUDIO OFF CACHE BOOL "" FORCE)
set(CYUI_BUILD_GALLERY OFF CACHE BOOL "" FORCE)
set(CYUI_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(CYUI_BUILD_WEB_CEF OFF CACHE BOOL "" FORCE)
add_subdirectory(${CYHUB_CYUI_SOURCE} ${CMAKE_CURRENT_BINARY_DIR}/cyui)
if(MSVC)
  target_compile_options(cyui_core PRIVATE "/FI${CMAKE_CURRENT_SOURCE_DIR}/ImGui192Compatibility.h")
else()
  target_compile_options(cyui_core PRIVATE -include "${CMAKE_CURRENT_SOURCE_DIR}/ImGui192Compatibility.h")
endif()
# One DLL, one Dear ImGui implementation/context API for both C# and CyUI.
add_library(cimgui SHARED ${CYHUB_CIMGUI_SOURCE}/cimgui.cpp CyUiBridge.cpp)
target_compile_definitions(cimgui PRIVATE CIMGUI_VARGS0)
target_link_libraries(cimgui PRIVATE cyhub_imgui CyUI::Core CyUI::DocumentViews CyUI::Diagnostics CyUI::Inspector CyUI::Images)
set_target_properties(cimgui PROPERTIES PREFIX "" RUNTIME_OUTPUT_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/runtime"
  LIBRARY_OUTPUT_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/runtime")
if(WIN32)
  target_link_libraries(cimgui PRIVATE imm32)
endif()
if(MSVC)
  target_compile_options(cyhub_imgui PRIVATE /utf-8)
  target_compile_options(cimgui PRIVATE /utf-8)
endif()
option(CYHUB_UI_TESTS "Build application integration contract tests" ON)
if(CYHUB_UI_TESTS)
  enable_testing()
  add_executable(cyhub_ui_tests CyUiBridgeTests.cpp)
  target_link_libraries(cyhub_ui_tests PRIVATE cimgui)
  target_include_directories(cyhub_ui_tests PRIVATE ${CYHUB_CIMGUI_SOURCE})
  target_compile_definitions(cyhub_ui_tests PRIVATE CIMGUI_DEFINE_ENUMS_AND_STRUCTS CIMGUI_NO_EXPORT IMGUI_USE_WCHAR32)
  add_test(NAME cyhub_ui_contract COMMAND cyhub_ui_tests)
  set_tests_properties(cyhub_ui_contract PROPERTIES WORKING_DIRECTORY "$<TARGET_FILE_DIR:cimgui>")
endif()
